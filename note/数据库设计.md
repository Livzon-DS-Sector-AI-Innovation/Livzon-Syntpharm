## 一、立项与开题模块

### ProjectInitiation 表

```sql
CREATE TABLE project_initiations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    project_id UUID NOT NULL REFERENCES rd_projects(id) ON DELETE CASCADE,
    
    -- 调研报告
    market_research TEXT,
    patent_analysis TEXT,
    sales_data JSONB,
    
    -- 立项报告
    development_strategy TEXT,
    target_product_profile JSONB,  -- QTPP
    
    -- 开题报告（QbD核心）
    synthetic_route TEXT,
    starting_materials JSONB,
    cqas JSONB,  -- 关键质量属性
    cpps JSONB,  -- 关键工艺参数
    
    -- 研究计划与预算
    gantt_chart JSONB,
    material_budget JSONB,
    labor_budget JSONB,
    total_cost_estimate NUMERIC(12, 2),
    
    status VARCHAR(20) DEFAULT 'draft',  -- draft/reviewed/approved
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    approved_at TIMESTAMP WITH TIME ZONE,
    approved_by UUID REFERENCES identity.users(id)
);

CREATE INDEX idx_initiations_project ON project_initiations(project_id);
CREATE INDEX idx_initiations_status ON project_initiations(status);
```

## 二、小试验证模块

### LabValidationBatch 表

```sql
CREATE TABLE lab_validation_batches (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    project_id UUID NOT NULL REFERENCES rd_projects(id) ON DELETE CASCADE,
    batch_no VARCHAR(50) NOT NULL,
    batch_sequence INT NOT NULL CHECK (batch_sequence IN (1, 2, 3)),
    
    -- 工艺参数记录
    process_parameters JSONB,
    yield_pct NUMERIC(5, 2),
    purity_pct NUMERIC(5, 2),
    
    -- 质量检测结果
    qc_results JSONB,
    impurity_profile JSONB,
    
    status VARCHAR(20) DEFAULT 'planned',  -- planned/in_progress/completed
    start_date DATE,
    end_date DATE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE UNIQUE INDEX idx_lab_batches_unique ON lab_validation_batches(project_id, batch_sequence);
CREATE INDEX idx_lab_batches_project ON lab_validation_batches(project_id);
```

### PreStabilityStudy 表

```sql
CREATE TABLE pre_stability_studies (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    project_id UUID NOT NULL REFERENCES rd_projects(id) ON DELETE CASCADE,
    batch_id UUID NOT NULL REFERENCES lab_validation_batches(id) ON DELETE CASCADE,
    
    study_type VARCHAR(30) NOT NULL,  -- forced_degradation/accelerated/long_term
    conditions JSONB,  -- {temperature: 40, humidity: 75}
    time_points INT[],  -- [0, 7, 14, 30, 60, 90]
    
    results JSONB,
    conclusion TEXT,
    
    status VARCHAR(20) DEFAULT 'planned',  -- planned/in_progress/completed
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_pre_stability_project ON pre_stability_studies(project_id);
CREATE INDEX idx_pre_stability_batch ON pre_stability_studies(batch_id);
```

## 三、中试研究模块

### PilotBatch 表

```sql
CREATE TABLE pilot_batches (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    project_id UUID NOT NULL REFERENCES rd_projects(id) ON DELETE CASCADE,
    batch_no VARCHAR(50) NOT NULL,
    scale_kg NUMERIC(10, 2),
    
    -- 生产记录
    production_log JSONB,
    equipment_used TEXT[],
    operators UUID[],  -- user IDs
    
    -- 质量结果
    yield_pct NUMERIC(5, 2),
    purity_pct NUMERIC(5, 2),
    qc_report_id UUID,
    
    status VARCHAR(20) DEFAULT 'planned',  -- planned/in_progress/completed
    start_date DATE,
    end_date DATE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_pilot_batches_project ON pilot_batches(project_id);
```

### CleaningProtocol 表

```sql
CREATE TABLE cleaning_protocols (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    project_id UUID NOT NULL REFERENCES rd_projects(id) ON DELETE CASCADE,
    equipment_id VARCHAR(50) NOT NULL,
    cleaning_method TEXT,
    acceptance_criteria JSONB,  -- {limit_ppm: 10, pde_value: 0.5}
    validation_status VARCHAR(20) DEFAULT 'draft',  -- draft/validated
    
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_cleaning_protocols_project ON cleaning_protocols(project_id);
```

### ProcessSpecification 表

```sql
CREATE TABLE process_specifications (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    project_id UUID NOT NULL REFERENCES rd_projects(id) ON DELETE CASCADE,
    version INT NOT NULL DEFAULT 1,
    content_md TEXT NOT NULL,
    critical_steps JSONB,
    parameters_range JSONB,
    
    is_current BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    created_by UUID REFERENCES identity.users(id)
);

CREATE INDEX idx_process_specs_project ON process_specifications(project_id);
CREATE UNIQUE INDEX idx_process_specs_current ON process_specifications(project_id) WHERE is_current = TRUE;
```

### BatchProductionRecord 表

```sql
CREATE TABLE batch_production_records (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    pilot_batch_id UUID NOT NULL REFERENCES pilot_batches(id) ON DELETE CASCADE,
    template_id UUID,  -- 关联空白模板
    
    filled_data JSONB,
    signatures UUID[],  -- user IDs who signed
    status VARCHAR(20) DEFAULT 'draft',  -- draft/completed/approved
    
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_batch_records_batch ON batch_production_records(pilot_batch_id);
```

## 五、任务协作模块

### ResearchTask 表

```sql
CREATE TABLE research_tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    project_id UUID NOT NULL REFERENCES rd_projects(id) ON DELETE CASCADE,
    
    title VARCHAR(200) NOT NULL,
    description TEXT,
    assignee_id UUID NOT NULL REFERENCES identity.users(id),
    due_date DATE,
    
    status VARCHAR(20) DEFAULT 'todo',  -- todo/in_progress/done
    priority VARCHAR(10) DEFAULT 'medium',  -- high/medium/low
    
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    completed_at TIMESTAMP WITH TIME ZONE
);

CREATE INDEX idx_tasks_project ON research_tasks(project_id);
CREATE INDEX idx_tasks_assignee ON research_tasks(assignee_id);
CREATE INDEX idx_tasks_status ON research_tasks(status);
```

## 八、数据字典

### 状态枚举值

| 字段 | 可选值 | 说明 |
|------|--------|------|
| ProjectInitiation.status | draft/reviewed/approved | 立项开题状态 |
| LabValidationBatch.status | planned/in_progress/completed | 小试批次状态 |
| PreStabilityStudy.status | planned/in_progress/completed | 预稳定性状态 |
| PilotBatch.status | planned/in_progress/completed | 中试批次状态 |
| CleaningProtocol.validation_status | draft/validated | 清洁方案状态 |
| BatchProductionRecord.status | draft/completed/approved | 批记录状态 |
| ImpuritySynthesisTrack.status | planning/synthesizing/confirmed | 杂质合成状态 |
| ResearchTask.status | todo/in_progress/done | 任务状态 |
| ResearchTask.priority | high/medium/low | 任务优先级 |
